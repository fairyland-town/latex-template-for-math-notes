\begin{definition}[Example]
  A map $f:X\to Y$ assigns a unique $y\in Y$ to each $x\in X$.
\end{definition}
\begin{theorem}[Intermediate Value Theorem]
  If $f$ is continuous on $[a,b]$, it attains every value between $f(a)$ and $f(b)$.
\end{theorem}
\begin{lemma}[Local Boundedness]
  If $f$ is continuous at $x_0$, it is bounded on some neighborhood of $x_0$.
\end{lemma}
\begin{corollary}[Boundedness]
  A continuous function on a closed interval is bounded.
\end{corollary}
\end{minipage}\hfill
\begin{minipage}[t]{0.485\linewidth}
\begin{criterion}[Cauchy Criterion]
  A sequence converges iff it is a Cauchy sequence.
\end{criterion}
\begin{proposition}[Uniqueness]
  A limit, if it exists, is unique.
\end{proposition}
\begin{example}[Shortcuts]
  Sets $\R,\Z,\N,\Q,\C$; norm $\norm{x}$; inner product $\inner{f}{g}$.
\end{example}
\begin{remark}[Note]
  Theorems, lemmas, corollaries and criteria share one numbering sequence.
\end{remark}
\end{minipage}
