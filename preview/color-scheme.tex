% 两列排版，左列四个、右列四个，底部放不编号的证明/解
\begin{minipage}[t]{0.485\linewidth}
\begin{definition}[示例]
  若 $f:X\to Y$ 对每个 $x\in X$ 有唯一 $y\in Y$ 对应，则称 $f$ 为映射。
\end{definition}
\begin{theorem}[介值定理]
  若 $f$ 在 $[a,b]$ 上连续，则 $f$ 取遍 $f(a)$ 与 $f(b)$ 之间的一切值。
\end{theorem}
\begin{lemma}[局部有界性]
  若 $f$ 在 $x_0$ 处连续，则 $f$ 在 $x_0$ 的某邻域内有界。
\end{lemma}
\begin{corollary}[有界性]
  闭区间上的连续函数必有界。
\end{corollary}
\end{minipage}\hfill
\begin{minipage}[t]{0.485\linewidth}
\begin{criterion}[柯西收敛准则]
  数列收敛当且仅当它是柯西列。
\end{criterion}
\begin{proposition}[唯一性]
  极限若存在则唯一。
\end{proposition}
\begin{example}[快捷命令]
  数集 $\R,\Z,\N,\Q,\C$；范数 $\norm{x}$；内积 $\inner{f}{g}$。
\end{example}
\begin{remark}[说明]
  定理、引理、推论、准则共用一套编号。
\end{remark}
\end{minipage}
